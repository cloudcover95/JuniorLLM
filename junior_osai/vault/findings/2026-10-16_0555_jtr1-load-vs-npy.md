## Question
Does the i2sd JTR1 load path cost less (time and peak RSS) than fp16/fp32 `.npy` load? Measure time and peak RSS to frame-decode a 16-layer JTR1 pack (stdlib `struct` header + numpy byte-table decode) against reading the same weights as fp16 or fp32 `.npy` from local disk. Load is memory-bound — packing might win even though matvec did not ([[2026-10-16_0255_jtr1-byte-lut-vs-fp32]]). This serves Quant Chief's pair_B item for this wave (JuniorOS overlay / bitnetd) and ROADMAP #3 i2sd userspace.

## Method
- python3, numpy 2.5.4, 8 CPUs, seed 0. No model, no GGUF, no network, nothing bound except named 127.0.0.1.
- Layer 4096x4096 exact ternary T in {-1,0,1} (uniform), gamma 0.02, L=16 layers. Write once under `/workspace/tbench/data/`: each layer as JTR1 frame, fp16 `.npy`, fp32 `.npy`.
- JTR1 layout from BitNet-mlx `scripts/i2sd_jtr1_pack.py`: magic `JTR1` + gamma f64 BE + n u32 BE + 2-bit I2_S MSB-first payload (0->0, 1->1, -1->2). Decode: read bytes, `struct.unpack`, 256x4 byte-table -> float32 weights.
- Timed loads: median of 11 repeats after 1 warm-up. Each path runs in a fresh child process so peak RSS (`resource.ru_maxrss` + `/proc/self/status` VmHWM) is isolated.
- Paths: (a) `np.load` fp32; (b) `np.load` fp16 keep; (c) `np.load` fp16 then upcast to fp32; (d) JTR1 frame-decode to float32. Also recorded packed-only read (bytes into RAM, no decode) for context.
- Assert JTR1 trit round-trip vs reference; rel err on decoded weights.
- Viable only if ternary ≥1.5x better on speed OR memory (peak RSS / on-disk bytes) with exact trit round-trip or rel err ≤1%.

### Command
`python3 /workspace/tbench/bench_jtr1_load_20261016.py` (local only). Full script:

```python
#!/usr/bin/env python3
"""JTR1 load-path bench vs fp16/fp32 .npy (wave 2026-10-16 05:05)."""
from __future__ import annotations
import gc, json, os, resource, struct, time, subprocess, sys
from pathlib import Path
import numpy as np

MAGIC = b"JTR1"
DEC = {0: 0, 1: 1, 2: -1, 3: 0}
ENC = np.array([2, 0, 1], np.uint8)
B2T = np.array([[DEC[(b >> s) & 3] for s in (6, 4, 2, 0)] for b in range(256)], np.float32)
M = K = 4096; L = 16; GAMMA = 0.02; SEED = 0
DATA = Path("/workspace/tbench/data"); N_WARM = 1; N_REP = 11; HEADER = 16

def pack_jtr1_payload(T):
    c = ENC[T + 1].reshape(T.shape[0], -1, 4)
    return ((c[..., 0] << 6) | (c[..., 1] << 4) | (c[..., 2] << 2) | c[..., 3]).astype(np.uint8)

def encode_jtr1_frame(T, gamma):
    n = int(T.size)
    return MAGIC + struct.pack(">dI", float(gamma), n) + pack_jtr1_payload(T).tobytes()

def decode_jtr1_to_weights(frame):
    gamma, n = struct.unpack(">dI", frame[4:HEADER])
    need = HEADER + (n * 2 + 7) // 8
    payload = np.frombuffer(frame[HEADER:need], dtype=np.uint8)
    return (np.float32(gamma) * B2T[payload].reshape(-1)[:n]).reshape(M, K)

def decode_jtr1_to_trits(frame):
    gamma, n = struct.unpack(">dI", frame[4:HEADER])
    need = HEADER + (n * 2 + 7) // 8
    payload = np.frombuffer(frame[HEADER:need], dtype=np.uint8)
    return B2T[payload].reshape(-1)[:n].astype(np.int8).reshape(M, K), float(gamma)

def peak_rss_mb():
    return resource.getrusage(resource.RUSAGE_SELF).ru_maxrss / 1024.0

def write_data():
    DATA.mkdir(parents=True, exist_ok=True)
    marker = DATA / "_ready.json"
    if marker.exists():
        meta = json.loads(marker.read_text())
        if meta.get("L") == L and (DATA / "jtr1_00.bin").exists():
            return meta
    rng = np.random.default_rng(SEED); g = np.float32(GAMMA)
    sizes = {"jtr1": 0, "fp16": 0, "fp32": 0}; ref_trits0 = None
    for i in range(L):
        T = rng.integers(-1, 2, (M, K), dtype=np.int8)
        if i == 0: ref_trits0 = T.copy()
        W = (g * T.astype(np.float32)).astype(np.float32)
        frame = encode_jtr1_frame(T, float(g))
        (DATA / f"jtr1_{i:02d}.bin").write_bytes(frame)
        np.save(DATA / f"fp16_{i:02d}.npy", W.astype(np.float16))
        np.save(DATA / f"fp32_{i:02d}.npy", W)
        sizes["jtr1"] += (DATA / f"jtr1_{i:02d}.bin").stat().st_size
        sizes["fp16"] += (DATA / f"fp16_{i:02d}.npy").stat().st_size
        sizes["fp32"] += (DATA / f"fp32_{i:02d}.npy").stat().st_size
        if i == 0:
            got, gg = decode_jtr1_to_trits(frame)
            assert np.array_equal(got, T) and abs(gg - float(g)) < 1e-15
        del T, W, frame; gc.collect()
    np.save(DATA / "ref_trits0.npy", ref_trits0)
    meta = {"L": L, "M": M, "K": K, "gamma": GAMMA, "sizes": sizes}
    marker.write_text(json.dumps(meta, indent=2)); return meta

def load_all_jtr1():
    return [decode_jtr1_to_weights((DATA / f"jtr1_{i:02d}.bin").read_bytes()) for i in range(L)]
def load_all_fp16(upcast=True):
    out = []
    for i in range(L):
        a = np.load(DATA / f"fp16_{i:02d}.npy")
        out.append(a.astype(np.float32) if upcast else a)
    return out
def load_all_fp32():
    return [np.load(DATA / f"fp32_{i:02d}.npy") for i in range(L)]

def child_bench(name, mode, out_path):
    fn = {"jtr1": load_all_jtr1, "fp16": lambda: load_all_fp16(True),
          "fp16_keep": lambda: load_all_fp16(False), "fp32": load_all_fp32}[mode]
    for _ in range(N_WARM):
        a = fn(); del a; gc.collect()
    ts = []; last = None
    for r in range(N_REP):
        gc.collect(); t0 = time.perf_counter(); last = fn(); ts.append((time.perf_counter() - t0) * 1e3)
        if r < N_REP - 1: del last; last = None; gc.collect()
    rss = peak_rss_mb(); vmhwm = None
    with open("/proc/self/status") as f:
        for line in f:
            if line.startswith("VmHWM:"): vmhwm = int(line.split()[1]) / 1024.0; break
    rel_err = trit_ok = None
    if mode == "jtr1" and last is not None:
        ref_T = np.load(DATA / "ref_trits0.npy")
        got_T, _ = decode_jtr1_to_trits((DATA / "jtr1_00.bin").read_bytes())
        trit_ok = bool(np.array_equal(got_T, ref_T))
        Wref = np.float32(GAMMA) * ref_T.astype(np.float32)
        rel_err = float(np.linalg.norm(last[0] - Wref) / (np.linalg.norm(Wref) + 1e-30))
    ts = np.array(ts)
    Path(out_path).write_text(json.dumps({
        "name": name, "mode": mode, "median_ms": float(np.median(ts)),
        "min_ms": float(ts.min()), "max_ms": float(ts.max()), "all_ms": [float(x) for x in ts],
        "peak_rss_mb_rusage": float(rss), "peak_rss_mb_vmhwm": float(vmhwm) if vmhwm else None,
        "rel_err": rel_err, "trit_ok": trit_ok, "n_layers": L, "pid": os.getpid()}, indent=2))
    return 0

def main():
    if len(sys.argv) > 1 and sys.argv[1] == "--child":
        return child_bench(sys.argv[2], sys.argv[3], sys.argv[4])
    meta = write_data(); sizes = meta["sizes"]
    results_dir = Path("/workspace/tbench/results"); results_dir.mkdir(parents=True, exist_ok=True)
    rows = []
    for name, mode in [("fp32_npy","fp32"),("fp16_npy_upcast","fp16"),("fp16_npy_keep","fp16_keep"),("jtr1_decode","jtr1")]:
        outp = str(results_dir / f"{mode}.json")
        p = subprocess.run([sys.executable, __file__, "--child", name, mode, outp], capture_output=True, text=True)
        if p.returncode: raise SystemExit(p.stderr or p.stdout)
        rows.append(json.loads(Path(outp).read_text()))
    base = next(r for r in rows if r["mode"] == "fp32")
    base_ms = base["median_ms"]; base_rss = base["peak_rss_mb_vmhwm"] or base["peak_rss_mb_rusage"]
    size_map = {"fp32": sizes["fp32"], "fp16": sizes["fp16"], "fp16_keep": sizes["fp16"], "jtr1": sizes["jtr1"]}
    print("| path | disk bytes | mem vs fp32 (disk) | median ms | min-max ms | speed vs fp32 load | peak RSS MB | RSS vs fp32 | rel err |")
    print("|---|---|---|---|---|---|---|---|---|")
    for r in rows:
        disk = size_map[r["mode"]]; rss = r["peak_rss_mb_vmhwm"] or r["peak_rss_mb_rusage"]
        err = r["rel_err"]; err_s = f"{err:.1e}" if err is not None else "n/a (npy)"
        print(f"| {r['name']} | {disk:,} | {sizes['fp32']/disk:.1f}x | {r['median_ms']:.2f} | "
              f"{r['min_ms']:.2f}-{r['max_ms']:.2f} | {base_ms/r['median_ms']:.2f}x | {rss:.1f} | "
              f"{base_rss/rss:.2f}x | {err_s} |")
    (results_dir / "summary.json").write_text(json.dumps({"sizes": sizes, "rows": rows, "base_ms": base_ms, "base_rss": base_rss}, indent=2))
    return 0

if __name__ == "__main__":
    raise SystemExit(main())
```

Context (not the scored path): packed-only read of all 16 JTR1 frames with no decode — median **40.2 ms**, peak RSS **~90 MB** (~10x faster and ~11.7x less RSS than fp32 load). That win disappears once numpy expands to float32.

JTR1 trit round-trip: `trit_ok=true`, rel err `0.0e+00` (exact).

## Verdict
**Not viable (load time and peak RSS after decode).** Frame-decoding 16 JTR1 layers to float32 is about **3.5x slower** than `np.load` of the matching fp32 `.npy` (1421 ms vs 405 ms) and peak RSS is essentially the same (~1123 MB vs ~1051 MB) because the decoded tensors are full fp32. Error is exact (0), and on-disk size is still **16.0x** smaller (67.1 MB vs 1073.7 MB), but that disk factor was already recorded in [[2026-10-16_0255_jtr1-byte-lut-vs-fp32]] / the matvec baseline — it is not a new load-path win. The open hypothesis ("load is memory-bound, packing might win") fails for the usable decode path in numpy: CPU byte-table expand dominates I/O savings when the page cache is warm, and peak RSS tracks the expanded fp32 footprint.

fp16 keep (no upcast) does clear the 1.5x bar on both speed (~1.7–2.1x) and RSS (~1.95x), but that is not ternary. Packed-resident JTR1 without decode would clear the bar hard (~10x speed, ~12x RSS vs fp32 load) — that is the shape bitnetd / a native I2_S kernel wants, not numpy dequant-on-load.
